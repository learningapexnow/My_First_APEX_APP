create or replace FUNCTION custom_auth (
  p_username IN VARCHAR2,
  p_password IN VARCHAR2
) RETURN BOOLEAN IS
  l_user_id         app_user.app_user_id%TYPE;
  l_hashed_password app_user.hashed_password%TYPE;
BEGIN
  -- Step 1: Find an active user with a matching username (case-insensitive).
  SELECT
    app_user_id,
    hashed_password
  INTO
    l_user_id,
    l_hashed_password
  FROM
    app_user
  WHERE
    upper(user_name) = upper(p_username)
    AND is_active_yn = 'Y';

  -- Step 2: Compare the provided password with the one stored in the table.
  IF p_password = l_hashed_password THEN
    -- Step 3 (Crucial): Passwords match. Set the security context for the session.
    -- This is the link that enables your secure views and permission checks.
    app_security.set_user_context(p_username);

    -- Step 4: Return TRUE to APEX, indicating a successful login.
    RETURN TRUE;
  ELSE
    -- Passwords do not match.
    RETURN FALSE;
  END IF;

EXCEPTION
  WHEN NO_DATA_FOUND THEN
    -- This block runs if the SELECT statement did not find a matching user.
    -- The user either does not exist or is not active. In either case, fail authentication.
    RETURN FALSE;
  WHEN OTHERS THEN
    -- Catch any other unexpected errors and fail gracefully.
    RETURN FALSE;
END custom_auth;
/

create or replace function sch_auth_simple_fn (
  p_username in varchar2,
  p_password in varchar2
) return boolean
is
  l_dummy number;
begin
  /*-----------------------------------------------------------------------------
    Simple Authentication Only
      - Validates that:
          1) the username exists in SCH_APP_USER
          2) the user is active (IS_ACTIVE_YN = 'Y')
          3) the stored PASSWORD matches the supplied password

    Notes:
      - This uses the SCH_APP_USER.PASSWORD column from your DDL.
      - No hashing, no lockout, no password policy (prototype only).
  -----------------------------------------------------------------------------*/

  select 1
    into l_dummy
    from sch_app_user u
   where upper(u.user_name)  = upper(p_username)
     and u.is_active_yn      = 'Y'
     and u.password          = p_password;

  return true;

exception
  when no_data_found then
    return false;
end sch_auth_simple_fn;
/

create or replace function sch_has_role1_fn (
  p_role_code in varchar2
)
  return boolean
is
  l_dummy number;
begin
  /*-----------------------------------------------------------------------------
    Returns TRUE if the current APEX session user:
      - exists in SCH_APP_USER and is active
      - has an active, effective-dated role matching p_role_code
        in SCH_APP_USER_ROLE
  -----------------------------------------------------------------------------*/
  select 1
    into l_dummy
    from sch_app_user u
    join sch_app_user_role ur
      on ur.app_user_id = u.app_user_id
   where upper(u.user_name) = upper(sys_context('APEX$SESSION','APP_USER'))
     and u.is_active_yn = 'Y'
     and upper(ur.role_code) = upper(p_role_code) -- Now uses the parameter
     and ur.is_active = 'Y'
     and (ur.effective_from is null or ur.effective_from <= trunc(sysdate))
     and (ur.effective_to   is null or ur.effective_to   >= trunc(sysdate))
     and rownum = 1; -- Optimization to stop searching after one match

  return true;

exception
  when no_data_found then
    return false;

end sch_has_role1_fn;
/
create or replace function sch_has_role_3tbl_fn (
  p_role in varchar2
) return boolean
is
  l_app_user_name  varchar2(255) := sys_context('APEX$SESSION','APP_USER');
  l_app_user_id    sch_app_user.app_user_id%type;
  l_role_code      sch_role.role_code%type;
  l_dummy          number;
begin
  select u.app_user_id
    into l_app_user_id
    from sch_app_user u
   where upper(u.user_name) = upper(l_app_user_name)
     and u.is_active_yn = 'Y';

  if regexp_like(trim(p_role), '^\d+$') then
    select r.role_code
      into l_role_code
      from sch_role r
     where r.role_id = to_number(trim(p_role));
  else
    select r.role_code
      into l_role_code
      from sch_role r
     where upper(r.role_code) = upper(trim(p_role));
  end if;

  select 1
    into l_dummy
    from sch_app_user_role ur
   where ur.app_user_id = l_app_user_id
     and upper(ur.role_code) = upper(l_role_code)
     and ur.is_active = 'Y'
     and (ur.effective_from is null or ur.effective_from <= trunc(sysdate))
     and (ur.effective_to   is null or ur.effective_to   >= trunc(sysdate));

  return true;
exception
  when no_data_found then
    return false;
end;
/

create or replace function sch_has_role_fn (
  p_role_code in varchar2
) return boolean
is
  l_dummy number;
begin
  /*-----------------------------------------------------------------------------
    Authorization-only check:
      - TRUE if current user has effective role p_role_code.
      - Relies on V_SCH_CURRENT_USER_ROLES, which already enforces:
          * SCH_APP_USER_ROLE.IS_ACTIVE = 'Y'
          * effective_from/effective_to window
  -----------------------------------------------------------------------------*/
  select 1 into l_dummy
    from v_sch_current_user_roles r
   where r.role_code = upper(p_role_code);

  return true;

exception
  when no_data_found then
    return false;
end sch_has_role_fn;
/
create or replace function sch_has_role_table_fn2 (
  p_role_code in varchar2
) return boolean
is
  l_dummy number;
begin
  /*-----------------------------------------------------------------------------
    Authorization-only check using tables directly:

    Steps:
      1) Find the current APEX user in SCH_APP_USER (active only)
      2) Confirm a role assignment exists in SCH_APP_USER_ROLE
         that is active and effective-dated
  -----------------------------------------------------------------------------*/
  select 1
    into l_dummy
    from sch_app_user u
    join sch_app_user_role ur
      on ur.app_user_id = u.app_user_id
   where upper(u.user_name) = upper(sys_context('APEX$SESSION','APP_USER'))
     and u.is_active_yn     = 'Y'
     and upper(ur.role_code)= upper(p_role_code)
     and ur.is_active       = 'Y'
     and (ur.effective_from is null or ur.effective_from <= trunc(sysdate))
     and (ur.effective_to   is null or ur.effective_to   >= trunc(sysdate));

  return true;

exception
  when no_data_found then
    return false;
end sch_has_role_table_fn2;

create or replace PROCEDURE grant_permission (
  p_role_code   IN VARCHAR2,
  p_perm_code   IN VARCHAR2,
  p_created_by  IN VARCHAR2 DEFAULT 'SEED_SCRIPT'
) IS
  l_log_message app_log.log_message%TYPE;
BEGIN -- This is the main procedure BEGIN

  -- Execute the main logic to grant the permission
  MERGE INTO app_role_permission rp
  USING (
    SELECT
      get_role_id(p_role_code) AS role_id,
      get_perm_id(p_perm_code) AS permission_id
    FROM dual
  ) src
  ON (rp.role_id = src.role_id AND rp.permission_id = src.permission_id)
  WHEN NOT MATCHED THEN
    INSERT (role_id, permission_id, created_by)
    VALUES (src.role_id, src.permission_id, p_created_by);

  -- Begin Logging Section
  IF SQL%ROWCOUNT > 0 THEN
    l_log_message := 'Permission granted. Role: ' || p_role_code || ', Permission: ' || p_perm_code;
  ELSE
    l_log_message := 'Permission already existed for Role: ' || p_role_code || ', Permission: ' || p_perm_code;
  END IF;

  -- Wrap the logging in its own transaction block.
  BEGIN -- This is the BEGIN for the inner logging block
    INSERT INTO app_log (log_level, log_message, log_context)
    VALUES ('SECURITY', l_log_message, 'grant_permission');
  EXCEPTION
    WHEN OTHERS THEN
      -- If logging fails, do nothing.
      NULL;
  END; -- THIS IS THE REQUIRED 'END;' FOR THE INNER LOGGING BLOCK

END grant_permission; -- THIS IS THE REQUIRED 'END;' FOR THE ENTIRE PROCEDURE;
/
create or replace PROCEDURE revoke_permission (
  p_role_code   IN VARCHAR2,
  p_perm_code   IN VARCHAR2
) IS
  l_log_message app_log.log_message%TYPE;
BEGIN
  -- Execute the main logic to revoke the permission.
  DELETE FROM app_role_permission
  WHERE role_id = get_role_id(p_role_code)
    AND permission_id = get_perm_id(p_perm_code);

  -- Begin Logging Section
  IF SQL%ROWCOUNT > 0 THEN
    l_log_message := 'Permission revoked. Role: ' || p_role_code || ', Permission: ' || p_perm_code;
  ELSE
    l_log_message := 'Revoke ignored. Permission did not exist for Role: ' || p_role_code || ', Permission: ' || p_perm_code;
  END IF;

  -- Use the same robust, self-contained logging block.
  BEGIN
    INSERT INTO app_log (log_level, log_message, log_context)
    -- CORRECTED LINE: The variable is now spelled correctly.
    VALUES ('SECURITY', l_log_message, 'revoke_permission');
  EXCEPTION
    WHEN OTHERS THEN
      NULL; -- Do not let a logging failure cause the main operation to fail.
  END;

END revoke_permission;

        FROM   v_sch_current_user cu

        AND    uss.student_id = m.student_id

create or replace PACKAGE BODY app_security AS

  PROCEDURE set_user_context(
    p_user_name IN VARCHAR2
  ) IS
  BEGIN
    dbms_session.set_identifier(p_user_name);
  END set_user_context;

  ----------------------------------------------------------------------------

  PROCEDURE clear_user_context IS
  BEGIN
    dbms_session.clear_identifier;
  END clear_user_context;

  ----------------------------------------------------------------------------

  FUNCTION has_role(
    p_role_code IN VARCHAR2
  ) RETURN BOOLEAN IS
    l_user_id   app_user.app_user_id%TYPE;
    l_role_count NUMBER;
  BEGIN
    BEGIN
      SELECT app_user_id INTO l_user_id FROM v_sch_current_user;
    EXCEPTION
      WHEN NO_DATA_FOUND THEN
        RETURN FALSE;
    END;

    SELECT COUNT(*)
      INTO l_role_count
      FROM app_user_role ur
      JOIN app_role r ON r.role_id = ur.role_id
     WHERE ur.app_user_id = l_user_id
       AND r.role_code = p_role_code;

    RETURN (l_role_count > 0);
  END has_role;
  
  ----------------------------------------------------------------------------
  
  FUNCTION is_admin RETURN BOOLEAN IS
  BEGIN
    RETURN has_role('ADMIN');
  END is_admin;

  ----------------------------------------------------------------------------

  FUNCTION has_permission(
    p_permission_code IN VARCHAR2
  ) RETURN BOOLEAN IS
    l_user_id   app_user.app_user_id%TYPE;
    l_perm_count NUMBER;
  BEGIN
    BEGIN
      SELECT app_user_id INTO l_user_id FROM v_sch_current_user;
    EXCEPTION
      WHEN NO_DATA_FOUND THEN
        RETURN FALSE;
    END;

    SELECT COUNT(*)
      INTO l_perm_count
      FROM app_role_permission rp
      JOIN app_permission p ON p.permission_id = rp.permission_id
      JOIN app_user_role ur ON ur.role_id = rp.role_id
     WHERE ur.app_user_id = l_user_id
       AND p.permission_code = p_permission_code;

    RETURN (l_perm_count > 0);
  END has_permission;

END app_security;
/
create or replace package body sch_audit_api as
  function current_user return varchar2 is
    l_user varchar2(255);
  begin
    l_user := coalesce(
                sys_context('APEX$SESSION','APP_USER'),
                sys_context('USERENV','CLIENT_IDENTIFIER'),
                sys_context('USERENV','SESSION_USER')
              );
    return lower(l_user);
  end current_user;
end sch_audit_api;

create or replace PACKAGE BODY sch_security_pkg AS

  FUNCTION is_authenticated RETURN BOOLEAN IS
    l_dummy NUMBER;
  BEGIN
    SELECT 1 INTO l_dummy
    FROM v_sch_current_user;
    RETURN TRUE;
  EXCEPTION
    WHEN NO_DATA_FOUND THEN
      RETURN FALSE;
    WHEN OTHERS THEN
      RETURN FALSE;
  END;

  FUNCTION has_role(p_role_code IN VARCHAR2) RETURN BOOLEAN IS
    l_dummy NUMBER;
  BEGIN
    IF p_role_code IS NULL THEN
      RETURN FALSE;
    END IF;

    SELECT 1 INTO l_dummy
    FROM v_sch_current_user_roles r
    WHERE UPPER(r.role_code) = UPPER(TRIM(p_role_code));

    RETURN TRUE;
  EXCEPTION
    WHEN NO_DATA_FOUND THEN
      RETURN FALSE;
    WHEN OTHERS THEN
      RETURN FALSE;
  END;

  FUNCTION has_any_role(p_role_codes_csv IN VARCHAR2) RETURN BOOLEAN IS
    l_cnt NUMBER;
  BEGIN
    IF p_role_codes_csv IS NULL THEN
      RETURN FALSE;
    END IF;

    SELECT COUNT(*)
      INTO l_cnt
    FROM v_sch_current_user_roles r
    WHERE EXISTS (
      SELECT 1
      FROM (
        SELECT TRIM(REGEXP_SUBSTR(p_role_codes_csv, '[^,]+', 1, LEVEL)) AS role_code
        FROM dual
        CONNECT BY REGEXP_SUBSTR(p_role_codes_csv, '[^,]+', 1, LEVEL) IS NOT NULL
      ) x
      WHERE UPPER(x.role_code) = UPPER(r.role_code)
    );

    RETURN (l_cnt > 0);
  EXCEPTION
    WHEN OTHERS THEN
      RETURN FALSE;
  END;

  FUNCTION is_admin     RETURN BOOLEAN IS BEGIN RETURN has_role('ADMIN');     END;
  FUNCTION is_registrar RETURN BOOLEAN IS BEGIN RETURN has_role('REGISTRAR'); END;
  FUNCTION is_teacher   RETURN BOOLEAN IS BEGIN RETURN has_role('TEACHER');   END;
  FUNCTION is_parent    RETURN BOOLEAN IS BEGIN RETURN has_role('PARENT');    END;
  FUNCTION is_student   RETURN BOOLEAN IS BEGIN RETURN has_role('STUDENT');   END;
  FUNCTION is_nurse     RETURN BOOLEAN IS BEGIN RETURN has_role('NURSE');     END;

END sch_security_pkg;

create or replace PACKAGE BODY sch_test_ctx_pkg AS

  ------------------------------------------------------------------------------
  -- set_user
  -- Sets the test user name into the SCH_TEST_CTX context for this session.
  -- Input is normalized to UPPER/TRIM to match common APEX username practice.
  ------------------------------------------------------------------------------
  PROCEDURE set_user(p_user_name IN VARCHAR2) IS
    l_user VARCHAR2(255);
  BEGIN
    l_user := UPPER(TRIM(p_user_name));

    IF l_user IS NULL THEN
      -- If you pass NULL/empty, treat it like clearing the override.
      clear_user;
      RETURN;
    END IF;

    DBMS_SESSION.set_context(
      namespace => 'SCH_TEST_CTX',
      attribute => 'TEST_USER_NAME',
      value     => l_user
    );
  END set_user;

  ------------------------------------------------------------------------------
  -- clear_user
  -- Clears the test user override for this session.
  ------------------------------------------------------------------------------
  PROCEDURE clear_user IS
  BEGIN
    DBMS_SESSION.set_context(
      namespace => 'SCH_TEST_CTX',
      attribute => 'TEST_USER_NAME',
      value     => NULL
    );
  END clear_user;

  ------------------------------------------------------------------------------
  -- get_user
  -- Returns the current override value (if any).
  ------------------------------------------------------------------------------
  FUNCTION get_user RETURN VARCHAR2 IS
  BEGIN
    RETURN SYS_CONTEXT('SCH_TEST_CTX', 'TEST_USER_NAME');
  END get_user;

END sch_test_ctx_pkg;
